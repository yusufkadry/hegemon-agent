survival_df = None
ih_dict = {'ArrayID':[], 'ArrayHeader':[], 'ClinicalHeader':[]}
num_samples = len(gse.gsms.items())
for i, (name, gsm) in enumerate(gse.gsms.items()):
    if print_progress:
        print(f"{name} \t {i+1}/{num_samples}") 
    gsm_meta = {}
    gsm_meta['ArrayId'] = name
    gsm_meta['time'] = ''
    gsm_meta['status'] = ''

    ih_dict['ArrayID'].append(name)
    ih_dict['ArrayHeader'].append(name)
    ih_dict['ClinicalHeader'].append(gsm.metadata['title'][0])

    remaining_cols = {}
    for col, val in gsm.metadata.items():
        used = False
        # Metadata columns
        if col == 'title' or '_ch' in col:
            
            # Only keeping cols that don't have protocol and label -> not needed
            if 'protocol' not in col and 'label' not in col and 'channel_count' not in col:
                used = True
                # Sometimes all metadata in characteristics column, so need to split (ex age: 17, gender: male)
                if ':' in val[0]:
                    for v in val:
                        # Split on ':' so everything before ':' is a new col and after ':' is the value
                        new_val = v.split(': ')
                        if len(new_val) == 2:
                            gsm_meta['c '+ new_val[0]+'_ch1'] = new_val[1]
                        else:
                            gsm_meta['c '+new_val[0][:-1]+'_ch1'] = 'N/A' 
                else:
                    gsm_meta['c '+ col] = val
         
        if not used:
            remaining_cols[col] = val
        
        # If need to add extra survival information
        # Ex:
        # meta_df['c description'] = meta_df['description']

    if survival_df is None:
        survival_df = pd.DataFrame(gsm_meta)
    else:
        temp_df = pd.DataFrame(gsm_meta)
        survival_df = survival_df.append(temp_df, sort=False)

## Generate idx file

===== CELL 31 (CODE) =====

ptr = []
ids = []
name = []
desc = []
pos = 0
expr_file_name = f'{accessionID}-{gpl.name}-expr.txt'
print('- Starting idx file generationn')
with open(expr_file_name, 'rb') as f:
    print('- Reading bytes from', expr_file_name)
    for line in f:
        if pos == 0:
            pos += len(line)
        else:
            ptr.append(pos)
            pos += len(line)
            split = line.decode("utf-8").split('\t')
            ids.append(split[0])
            name.append(split[1].split(':')[0])
            desc.append(':'.join(split[1].split(':')[1:]))
    f.close()

with open(f'{accessionID}-{gpl.name}-idx.txt', 'w') as f:
    print('- Writing idx file')
    f.write('ProbeID\tPtr\tName\tDescription\n')
    for i in range(len(ids)):
        f.write('{}\t{}\t{}\t{}\n'.format(ids[i], ptr[i], name[i], desc[i]))
    f.close()
print('- Completed')
